> iqr_outliers_toNA <- function(x, range = 5, bound = "both") {
+     upperBound = mean(x, na.rm = T) + IQR(x, na.rm = T) * range
+     lowerBound = mean(x, na.rm = T) - IQR(x, na.rm = T) * range
+     print(glue::glue("Upper outliers:{paste(x[x > upperBound],' ',collapse='')}"))
+     print(glue::glue("Lower outliers:{paste(x[x < lowerBound],' ',collapse='')}"))
+     if (bound == "both") 
+         x[x > upperBound | x < lowerBound] = NA
+     if (bound == "lower") 
+         x[x < lowerBound] = NA
+     if (bound == "upper") 
+         x[x > upperBound] = NA
+     return(x)
+ }

> calcSMR = function(Y, q = c(0.1, 0.15, 0.2, 0.25, 
+     0.3), G = 1:4) {
+     require(mclust)
+     u = sort(Y)
+     the.Mclust <- Mclust(Y, G = G, verbose = F)
+     cl <- the.Mclust$classification
+     cl2 <- as.data.frame(table(cl))
+     cl2$cl <- as.numeric(levels(cl2$cl))
+     valid <- cl2$Freq >= 0.1 * length(time)
+     the.cl <- min(cl2$cl[valid])
+     left.distr <- Y[the.Mclust$classification == the.cl]
+     mlnd = the.Mclust$parameters$mean[the.cl]
+     CVmlnd = sd(left.distr)/mlnd * 100
+     quant = quantile(Y, q)
+     low10 = mean(u[1:10])
+     low10pc = mean(u[6:(5 + round(0.1 * (length(u) - 5)))])
+     return(list(mlnd = mlnd, quant = quant, low10 = low10, low10pc = low10pc, 
+         cl = cl, CVmlnd = CVmlnd))
+ }

> f_avgTemp <- function(tb_temp, date_start, date_end, 
+     SD = F) {
+     temps <- tb_temp %>% filter(dateTime > date_start & dateTime < 
+         date_end) %>% pull(temp)
+     if (SD == T) 
+         return(meanSD(temps))
+     else return(mean(temps, na.rm = T))
+ }

> sem <- function(x) sqrt(var(x, na.rm = T)/(sum(!is.na(x))))

> meanSem <- function(x, dec_mean = 2, dec_spread = 2) {
+     require(glue)
+     .mean <- round(mean(x, na.rm = T), dec_mean)
+     .sem <- round(sem(x), dec_spread)
+     .n <- length(x[!is.na(x)])
+     glue("{.mean}±{.sem} (±SE, n={.n})")
+ }

> meanSD <- function(x, dec_mean = 2, dec_spread = 2) {
+     require(glue)
+     .mean <- round(mean(x, na.rm = T), dec_mean)
+     .sd <- round(sd(x), dec_spread)
+     .n <- length(x[!is.na(x)])
+     glue("{.mean}±{.sd} (±SD, n={.n})")
+ }

> sim_formula <- y ~ b0 + b1 * b2

> sim_model <- function(b0, b1, b2 = 0, c_intercept = 1, 
+     c_b0 = 0, c_b1 = 0, c_b2 = 0, c_b01 = 0, c_b02 = 0, c_b12 = 0, 
+     c_b123 = 0, sd = 1) {
+     result <- c_intercept + b0 * c_b0 + b1 * c_b1 + b2 * c_b2 + 
+         b0 * b1 * c_b01 + b0 * b2 * c_b02 + b1 * b2 * c_b12 + 
+         b0 * b1 * b2 * c_b123 + rnorm(n = length(b0), sd = sd, 
+         mean = 0)
+ }

> sim_dataset <- function(n) {
+     table <- data.frame(b0 = rnorm(n = n, mean = 1.07, sd = 0.3), 
+         b1 = c(rep(0, n/2), rep(1, n/2)), b2 = c(0, 1))
+     return(table)
+ }

> sim_power <- function(n, c_intercept = 1, c_b0 = 0.83, 
+     c_b1 = 0.04, c_b2 = 0, c_b01 = 0, c_b02 = 0, c_b12 = 0, c_b123 = 0, 
+     sd = 0.06, coeff = 20, simuls = 500) {
+     n_success = 0
+     for (i in 1:simuls) {
+         tb <- sim_dataset(n) %>% mutate(y = sim_model(b0, b1, 
+             b2, c_intercept, c_b0, c_b1, c_b2, c_b01, c_b02, 
+             c_b12, c_b123, sd))
+         pval = summary(lm(sim_formula, data = tb))$coefficients[coeff]
+         if (pval < 0.05) 
+             sign = T
+         else sign = F
+         if (sign) 
+             n_success = n_success + 1
+     }
+     return(n_success/simuls)
+ }
